class DynamicTokenCountRegularization(BaseRegularization):
    """Keep the learned number of time tokens near the preregistered target count."""

    def __init__(self, cfg: Config, weight: float = 1.0):
        super().__init__(cfg, name="dynamic_token_count", weight=weight)

    def forward(self, prediction: Dict[str, torch.Tensor], ground_truth: Dict[str, torch.Tensor],
                other_model_data: Dict[str, torch.Tensor]) -> torch.Tensor:
        if "dynamic_token_count_loss" not in other_model_data:
            raise ValueError(
                "dynamic_token_count regularization requires model output 'dynamic_token_count_loss'.")
        # Evaluation keeps a length-one model output so the tester can concatenate diagnostics across batches.
        # Loss regularizers, however, must be scalar because they are added to a scalar prediction loss.
        return other_model_data["dynamic_token_count_loss"].mean()
